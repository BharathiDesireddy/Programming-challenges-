# problem no : 1886
# problem : [determine whether the matrix can be obtained by rotation](https://leetcode.com/problems/determine-whether-matrix-can-be-obtained-by-rotation/description/)
# Description : 
Given two n x n binary matrices mat and target, return true if it is possible to make mat equal to target by rotating mat in 90-degree increments, or false otherwise.

 

Example 1:


Input: mat = [[0,1],[1,0]], target = [[1,0],[0,1]]

Output: true

Explanation: We can rotate mat 90 degrees clockwise to make mat equal target.

### java Solution :
```java
class Solution {
    public boolean findRotation(int[][] mat, int[][] target) {
        // Check all 4 possible rotations (0, 90, 180, 270 degrees)
        for (int i = 0; i < 4; i++) {
            if (compareMatrices(mat, target)) {
                return true;
            }
            rotateMatrix(mat); // Rotate matrix 90 degrees clockwise
        }
        return false; // If no match found after 4 rotations
    }

    // Function to compare two matrices
    private boolean compareMatrices(int[][] mat, int[][] target) {
        for (int i = 0; i < mat.length; i++) {
            for (int j = 0; j < mat[i].length; j++) {
                if (mat[i][j] != target[i][j]) {
                    return false;
                }
            }
        }
        return true;
    }

    // Function to rotate the matrix 90 degrees clockwise
    private void rotateMatrix(int[][] matrix) {
        // Transpose the matrix
        transposeMatrix(matrix);
        // Reverse each row
        for (int[] row : matrix) {
            rowReverse(row);
        }
    }

    // Function to transpose the matrix
    private void transposeMatrix(int[][] matrix) {
        for (int i = 0; i < matrix.length; i++) {
            for (int j = i + 1; j < matrix[i].length; j++) {
                int temp = matrix[i][j];
                matrix[i][j] = matrix[j][i];
                matrix[j][i] = temp;
            }
        }
    }

    // Function to reverse a row
    private void rowReverse(int[] row) {
        int left = 0, right = row.length - 1;
        while (left < right) {
            int temp = row[left];
            row[left] = row[right];
            row[right] = temp;
            left++;
            right--;
        }
    }
}
