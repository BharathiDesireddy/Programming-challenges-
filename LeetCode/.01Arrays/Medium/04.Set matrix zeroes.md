# problem no : 73
# problem : [set matrix zeroes](https://leetcode.com/problems/set-matrix-zeroes/description/)
# Description : 
Given an m x n integer matrix matrix, if an element is 0, set its entire row and column to 0's.

You must do it in place.

 

Example 1:


Input: matrix = [[1,1,1],[1,0,1],[1,1,1]]

Output: [[1,0,1],[0,0,0],[1,0,1]]

# Java Solution : 
```java
class Solution {
    public void setZeroes(int[][] matrix) {
        boolean fr = false, fc = false;
        
        // Check if the first row needs to be zeroed
        for (int j = 0; j < matrix[0].length; j++) {
            if (matrix[0][j] == 0) {
                fr = true;
                break;
            }
        }
        
        // Check if the first column needs to be zeroed
        for (int i = 0; i < matrix.length; i++) {
            if (matrix[i][0] == 0) {
                fc = true;
                break;
            }
        }
        
        // Use the first row and first column to mark rows and columns
        // that should be zeroed
        for (int i = 1; i < matrix.length; i++) {
            for (int j = 1; j < matrix[0].length; j++) {
                if (matrix[i][j] == 0) {
                    matrix[i][0] = 0;  // Mark the first column
                    matrix[0][j] = 0;  // Mark the first row
                }
            }
        }
        
        // Zero out cells based on the marks in the first row and first column
        for (int i = 1; i < matrix.length; i++) {
            for (int j = 1; j < matrix[0].length; j++) {
                if (matrix[i][0] == 0 || matrix[0][j] == 0) {
                    matrix[i][j] = 0;
                }
            }
        }
        
        // Zero out the first row if needed
        if (fr) {
            for (int j = 0; j < matrix[0].length; j++) {
                matrix[0][j] = 0;
            }
        }
        
        // Zero out the first column if needed
        if (fc) {
            for (int i = 0; i < matrix.length; i++) {
                matrix[i][0] = 0;
            }
        }
    }
}
